Output_Tester_Position_PWM_LF ()
{
    word temp;
    float duty;
  
    if (@System::Position_Override_Enable::LF == TRUE)
    {
      duty = @System::Position_Override::LF;
      if (duty < 0.0) duty = 0.0;
      if (duty > 100.0) duty = 100.0;
      if (@Vehicle_Specific::Position_Invert[LF_CORNER] == TRUE) duty = (100.0 - duty);
    }
    else
    {
      /* Position_Display is 0-100 */
      if (Position_DC_Cmd[LF_CORNER] < 3) 
      {
         Position_DC_Cmd[LF_CORNER] = 3;
      }
      else if (Position_DC_Cmd[LF_CORNER] > 97)
      {
        Position_DC_Cmd[LF_CORNER] = 97;
      }
      
      duty  = Position_DC_Cmd[LF_CORNER];
    }
    @Display::Position::LF = duty;
    temp = duty*327.68;
    tester_cmd.Tester_PWM_DC_LF = temp;
}

Output_IAMM_Temperature ()
{
    float voltage; // 0-5 volt -40 (90%Vpwr) to 125 (10%Vpwr); 

    //voltage = 5-(((4.0/165.0)*(IAMM_Temperature + 40)) + 0.5);
    voltage = (-0.8/165.0 * (IAMM_Temperature + 40.0) + 0.9) * c_Tester_Temperature_Max_Voltage;
	  tester_cmd.Tester_Temperature = voltage*4095/5;
}

Override_Manifold_Pressure()
{
  if (TRUE == @User::Manifold_Pressure_Override_Enable)
  {
    Final_Manifold_Pressure = Manual_Manifold_Pressure;
  }
  else
  {
    Final_Manifold_Pressure = Manifold_Pressure;
  }
}

Output_Tester_Manifold_Pressure_P1 ()
{
    float voltage; // 0-5 volt
  
    voltage = (0.8/20 * Final_Manifold_Pressure + 0.1) * c_Tester_Temperature_Max_Voltage;
  
    if (voltage < 0)
    {
        voltage = 0.0;   
    }    
    else if (voltage > 5)
    {
        voltage = 5.0;
    }
    tester_cmd.Tester_Pressure_P1 = voltage*4095/5;   //12 Bit
}

Output_Tester_Manifold_Pressure_P2()
{
    int temp;
    double voltage; // 0-5 volt
  
    voltage = (-(0.8/20) * Final_Manifold_Pressure + 0.9) * c_Tester_Manifold_Pressure_Max_Voltage;

    if (voltage < 0)
    {
        voltage = 0.0;   
    }    
    else if (voltage > 5)
    {
        voltage = 5.0;
    }
    
    if (@Fault::Pressure_Correlation_Fault == TRUE)     //add 1V to P2 
    {
      if ((voltage + 1) <= 5) 
      {
        voltage = voltage + 1;
      }
      else  
      {
        voltage = voltage - 1;
      }
    }
    tester_cmd.Tester_Pressure_P2 = voltage*4095/5;   //12 Bit
}

Determine_PWM_DC()
{
  int i;
  int axle_index;

  axle_index = 0;
  
  for (i = LF_CORNER; i <= RR_CORNER; i++)
  {
      if (i >= LR_CORNER) 
      {
        axle_index = 1;
      }
      if (TRUE == @Vehicle_Specific::Position_Invert[i])
      {
          Position_DC_Cmd[i] = 100 - (@Vehicle_Specific::k_Position_PWM_Target[i] +
              ((@Vehicle_Specific::Spring_Position[i] - Nominal_Trim_Height[axle_index]) / 
                @Vehicle_Specific::k_Trim_Height_Gain[axle_index]));
      }
      else
      {
          Position_DC_Cmd[i] = @Vehicle_Specific::k_Position_PWM_Target[i] +
              ((@Vehicle_Specific::Spring_Position[i] - Nominal_Trim_Height[axle_index]) / 
                @Vehicle_Specific::k_Trim_Height_Gain[axle_index]);
		  if (Position_DC_Cmd[i] > 100) Position_DC_Cmd[i] = 100;
		  if (Position_DC_Cmd[i] < 0) Position_DC_Cmd[i] = 0;
      }
  }
}

on sysvar sysvar::Vehicle::Spring_Pressures_Offset_Enable
{
  if (@sysvar::Vehicle::Spring_Pressures_Offset_Enable == 1)
  {
    if ((@sysvar::Vehicle::Spring_Pressures_Offset[0] > 0) || (@sysvar::Vehicle::Spring_Pressures_Offset[1] > 0) ||
        (@sysvar::Vehicle::Spring_Pressures_Offset[2] > 0) || (@sysvar::Vehicle::Spring_Pressures_Offset[3] > 0))
    {
       @sysvar::System::Leak_Rate = abs(@sysvar::System::Leak_Rate);
    }
    else
    {
      @sysvar::System::Leak_Rate = -abs(@sysvar::System::Leak_Rate);
    }
    if (@sysvar::Vehicle::Spring_Pressures_Offset[0] != 0)
    {
      @sysvar::Vehicle::Spring_Pressures_Offset_Target[0] = @sysvar::Vehicle::Spring_Pressures[0] + @sysvar::Vehicle::Spring_Pressures_Offset[0];
      @sysvar::System::LF_Leak_Enable = 1;
      setTimer (Timer_LF_Offset, 10);
      
    }
    if (@sysvar::Vehicle::Spring_Pressures_Offset[1] != 0)
    {
      @sysvar::Vehicle::Spring_Pressures_Offset_Target[1] = @sysvar::Vehicle::Spring_Pressures[1] + @sysvar::Vehicle::Spring_Pressures_Offset[1];
      @sysvar::System::RF_Leak_Enable = 1;
      setTimer (Timer_RF_Offset, 10);
    }
    if (@sysvar::Vehicle::Spring_Pressures_Offset[2] != 0)
    {
